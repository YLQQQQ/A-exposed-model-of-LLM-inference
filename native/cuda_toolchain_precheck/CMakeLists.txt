cmake_minimum_required(VERSION 3.31)
project(native_cuda_link_check LANGUAGES CXX CUDA)
if(NOT WIN32 OR NOT MSVC OR NOT CMAKE_CXX_COMPILER_VERSION STREQUAL "19.38.33135")
  message(FATAL_ERROR "Requires Windows x64 pinned MSVC 19.38.33135")
endif()
if(NOT CMAKE_CUDA_COMPILER_VERSION STREQUAL "12.4.131" OR NOT CMAKE_CUDA_ARCHITECTURES STREQUAL "89")
  message(FATAL_ERROR "Requires existing CUDA 12.4.131 and explicit sm89")
endif()
find_package(CUDAToolkit 12.4 REQUIRED)
add_executable(native-cuda-link-check main.cu)
target_link_libraries(native-cuda-link-check PRIVATE CUDA::cudart)
target_compile_features(native-cuda-link-check PRIVATE cxx_std_17 cuda_std_17)
