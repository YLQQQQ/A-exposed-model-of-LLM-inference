cmake_minimum_required(VERSION 3.24)
project(exposedpath_native_llama LANGUAGES C CXX)
if(NOT WIN32 OR NOT MSVC)
  message(FATAL_ERROR "Candidate target requires Windows/VS2022; no fallback backend")
endif()
if(NOT DEFINED LLAMA_SOURCE_ROOT OR NOT DEFINED NATIVE_PATCHED_BACKEND OR NOT DEFINED NVTX_INCLUDE_ROOT)
  message(FATAL_ERROR "Explicit pinned source, patched backend and NVTX include paths required")
endif()
find_package(Git REQUIRED)
if(NOT IS_DIRECTORY "${LLAMA_SOURCE_ROOT}/.git/refs" OR NOT IS_DIRECTORY "${LLAMA_SOURCE_ROOT}/.git/objects" OR NOT EXISTS "${LLAMA_SOURCE_ROOT}/.git/HEAD")
  message(FATAL_ERROR "Pinned vendor Git structure missing; no parent discovery")
endif()
execute_process(COMMAND "${GIT_EXECUTABLE}" "--git-dir=${LLAMA_SOURCE_ROOT}/.git" "--work-tree=${LLAMA_SOURCE_ROOT}" rev-parse HEAD OUTPUT_VARIABLE base OUTPUT_STRIP_TRAILING_WHITESPACE RESULT_VARIABLE rc)
if(NOT rc EQUAL 0 OR NOT base STREQUAL "d81235049384534c167caea52b85a694f6103d14")
  message(FATAL_ERROR "llama.cpp source commit conflict")
endif()
execute_process(COMMAND "${GIT_EXECUTABLE}" "--git-dir=${LLAMA_SOURCE_ROOT}/.git" "--work-tree=${LLAMA_SOURCE_ROOT}" status --porcelain OUTPUT_VARIABLE dirty RESULT_VARIABLE rc)
if(NOT rc EQUAL 0 OR NOT dirty STREQUAL "")
  message(FATAL_ERROR "Pinned vendor source must remain clean")
endif()
file(SHA256 "${LLAMA_SOURCE_ROOT}/ggml/src/ggml-cuda/ggml-cuda.cu" original_sha)
file(SHA256 "${LLAMA_SOURCE_ROOT}/vendor/nlohmann/json.hpp" json_sha)
if(NOT original_sha STREQUAL "1315f06baff63da74e085b7ec01974474dac1f8ec1074af13eb70faba4751961" OR
   NOT json_sha STREQUAL "aaf127c04cb31c406e5b04a63f1ae89369fccde6d8fa7cdda1ed4f32dfc5de63")
  message(FATAL_ERROR "Upstream byte conflict (LF checkout required for pinned source)")
endif()
find_package(CUDAToolkit 12.4 REQUIRED)
if(NOT CUDAToolkit_VERSION MATCHES "^12\\.4\\.")
  message(FATAL_ERROR "No Toolkit upgrade or fallback authorized")
endif()
if(CMAKE_GENERATOR STREQUAL "Ninja")
  if(NOT CMAKE_C_COMPILER_VERSION STREQUAL "19.38.33135" OR NOT CMAKE_CXX_COMPILER_VERSION STREQUAL "19.38.33135")
    message(FATAL_ERROR "Actual MSVC 19.38.33135 required; no compiler fallback")
  endif()
elseif(CMAKE_GENERATOR STREQUAL "Visual Studio 17 2022")
  if(NOT CMAKE_VS_PLATFORM_TOOLSET_VERSION STREQUAL "14.38.33130")
    message(FATAL_ERROR "Explicit MSVC 14.38.33130 required")
  endif()
else()
  message(FATAL_ERROR "Unsupported generator; no fallback")
endif()
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CUDA_ARCHITECTURES 89 CACHE STRING "Fixed RTX4090 domain" FORCE)
set(BUILD_SHARED_LIBS OFF CACHE BOOL "One observer instance" FORCE)
set(GGML_CUDA ON CACHE BOOL "CUDA only" FORCE)
set(GGML_CUDA_GRAPHS OFF CACHE BOOL "Qualified eager candidate only" FORCE)
set(GGML_NATIVE OFF CACHE BOOL "Fixed source/binary receipt" FORCE)
set(LLAMA_BUILD_TESTS OFF CACHE BOOL "Minimal target" FORCE)
set(LLAMA_BUILD_EXAMPLES OFF CACHE BOOL "Minimal target" FORCE)
set(LLAMA_BUILD_TOOLS OFF CACHE BOOL "Minimal target" FORCE)
add_library(exposedpath-native-observer STATIC observer.cpp)
target_include_directories(exposedpath-native-observer PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}" "${LLAMA_SOURCE_ROOT}/vendor")
target_link_libraries(exposedpath-native-observer PUBLIC bcrypt)
add_subdirectory("${LLAMA_SOURCE_ROOT}" llama-build)
# Replace precisely one build input. Original vendor source is never rewritten.
get_target_property(cuda_sources ggml-cuda SOURCES)
set(replacement_count 0)
set(adjusted_sources "")
foreach(item IN LISTS cuda_sources)
  get_filename_component(name "${item}" NAME)
  if(name STREQUAL "ggml-cuda.cu")
    math(EXPR replacement_count "${replacement_count}+1")
    list(APPEND adjusted_sources "${NATIVE_PATCHED_BACKEND}")
  else()
    list(APPEND adjusted_sources "${item}")
  endif()
endforeach()
if(NOT replacement_count EQUAL 1)
  message(FATAL_ERROR "CUDA source list differs; no silent patch fallback")
endif()
set_property(TARGET ggml-cuda PROPERTY SOURCES "${adjusted_sources}")
target_include_directories(ggml-cuda PRIVATE "${LLAMA_SOURCE_ROOT}/ggml/src/ggml-cuda")
target_link_libraries(ggml-cuda PRIVATE exposedpath-native-observer)
add_executable(native-llama main.cpp)
target_include_directories(native-llama PRIVATE "${NVTX_INCLUDE_ROOT}")
target_link_libraries(native-llama PRIVATE llama exposedpath-native-observer CUDA::cudart)
