cmake_minimum_required(VERSION 3.31)
project(native_toolchain_check LANGUAGES C CXX)
if(NOT WIN32 OR NOT MSVC OR NOT MSVC_VERSION EQUAL 1938)
  message(FATAL_ERROR "Requires the existing Windows x64 MSVC 19.38 toolchain")
endif()
if(CMAKE_GENERATOR STREQUAL "Ninja")
  if(NOT CMAKE_C_COMPILER_VERSION STREQUAL "19.38.33135" OR NOT CMAKE_CXX_COMPILER_VERSION STREQUAL "19.38.33135")
    message(FATAL_ERROR "Actual MSVC 19.38.33135 required")
  endif()
elseif(CMAKE_GENERATOR STREQUAL "Visual Studio 17 2022")
  if(NOT CMAKE_VS_PLATFORM_TOOLSET_VERSION STREQUAL "14.38.33130")
    message(FATAL_ERROR "Pinned MSVC tools 14.38.33130 required")
  endif()
else()
  message(FATAL_ERROR "Unsupported generator; no fallback")
endif()
add_executable(native-toolchain-check main.cpp helper.c)
target_compile_features(native-toolchain-check PRIVATE cxx_std_17)
